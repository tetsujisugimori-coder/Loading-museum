export const __clock = process.hrtime.bigint.bind(process.hrtime);
let records: Array<{label:string;ms:number}> = [];
export function __mark(label:string, start:bigint) { records.push({label, ms:Number(__clock()-start)/1e6}); }
function measure<T>(label:string, f:()=>T):T { const start=__clock(); try{return f();}finally{__mark(label,start);} }
