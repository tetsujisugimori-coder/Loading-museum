import { afterEach, __clock, __mark } from './runtime-profile';
// @vitest-environment jsdom
import { act, cleanup, fireEvent, render } from './runtime-profile';
import { beforeEach, expect, it, vi } from 'vitest';
import { ExpandedGamePreview } from '../app/components/WaapiExpandedPreviews';
import { waapiSamples } from '../app/data/waapiSamples';
beforeEach(() => { vi.useFakeTimers(); vi.spyOn(performance, 'now').mockImplementation(() => Date.now()); vi.setSystemTime(1000); Object.defineProperty(HTMLElement.prototype, 'animate', { configurable: true, value: vi.fn(() => ({ playState: 'running', cancel: vi.fn(function(this: {playState:string}) {this.playState='idle';}) })) }); });
afterEach(() => { cleanup(); vi.useRealTimers(); vi.restoreAllMocks(); });
function setup(kind: string, reduced=false) { const sample=waapiSamples.find(s=>s.render===kind)!; const r=render(<ExpandedGamePreview sample={sample} active reducedMotion={reduced}/>); return {...r, change:(active:boolean,motion:boolean)=>r.rerender(<ExpandedGamePreview sample={sample} active={active} reducedMotion={motion}/>), output:()=>r.container.querySelector('output')!.textContent!, click:(name:string)=>fireEvent.click(r.getByRole('button',{name}))}; }
it.each(['parry','dodge'])('%s reduced-motion labels agree with early and active judgment', kind=> {const r=setup(kind,true); r.click(kind==='parry'?'ENEMY ATTACK':'ATTACK START'); act(()=>vi.advanceTimersByTime(30)); expect(r.output()).toBe('WARNING'); act(()=>vi.advanceTimersByTime(390)); expect(r.output()).toBe('ACTIVE WINDOW'); r.click(kind==='parry'?'PARRY':'DODGE LEFT'); expect(r.output()).toMatch(/PERFECT PARRY|S-RANK/);});
it.each(['parry','dodge'])('%s ceases active label after success window',kind=>{const r=setup(kind); r.click(kind==='parry'?'ENEMY ATTACK':'ATTACK START'); act(()=>vi.advanceTimersByTime(681)); expect(r.output()).toContain('LATE');});
it('battle offscreen interruption returns FIELD and stays retryable',()=>{const r=setup('battle-transition');r.click('ENCOUNTER');act(()=>vi.advanceTimersByTime(500));r.change(false,false);act(()=>vi.advanceTimersByTime(1400));r.change(true,false);expect(r.container.querySelector('.gameBattle')!.className).toContain('sceneFIELD');expect(r.output()).toContain('INTERRUPTED');r.click('ENCOUNTER');act(()=>vi.advanceTimersByTime(950));expect(r.container.querySelector('.gameBattle')!.className).toContain('sceneBATTLE');expect(r.output()).toBe('READY');});
it('combat paused animations cancel on reduced motion without discarding damage',()=>{const r=setup('combat');r.click('CRITICAL');const animations=vi.mocked(HTMLElement.prototype.animate).mock.results.map(x=>x.value);animations.forEach(a=>a.playState='paused');r.change(true,true);expect(animations.every(a=>a.playState==='idle')).toBe(true);expect(r.output()).toBe('CRITICAL -999');});
it.each([false,true])('parry exact boundaries and adjacent times, reduced=%s',reduced=>{const r=setup('parry',reduced);for(const [elapsed,expected] of [[419,'EARLY · HIT'],[420,'PERFECT PARRY'],[680,'PERFECT PARRY'],[681,'LATE · HIT']] as const){r.click('RESET');r.click('ENEMY ATTACK');act(()=>vi.advanceTimersByTime(elapsed));r.click('PARRY');expect(r.output()).toBe(expected);}});
it.each([false,true])('dodge exact boundaries and adjacent times, reduced=%s',reduced=>{const r=setup('dodge',reduced);for(const [elapsed,expected] of [[419,'EARLY · HIT'],[420,'LEFT · S-RANK'],[680,'LEFT · S-RANK'],[681,'LATE · HIT']] as const){r.click('RESET');r.click('ATTACK START');act(()=>vi.advanceTimersByTime(elapsed));r.click('DODGE LEFT');expect(r.output()).toBe(expected);}});
it.each(['parry','dodge'])('%s mid-motion switch preserves clock; reset rejects old timer and supports zero start',kind=>{vi.setSystemTime(0);const r=setup(kind);const start=kind==='parry'?'ENEMY ATTACK':'ATTACK START';const action=kind==='parry'?'PARRY':'DODGE RIGHT';r.click(start);act(()=>vi.advanceTimersByTime(200));r.change(true,true);act(()=>vi.advanceTimersByTime(220));expect(r.output()).toBe('ACTIVE WINDOW');r.change(true,false);r.click(action);expect(r.output()).toMatch(/PERFECT PARRY|S-RANK/);r.click(start);r.click('RESET');act(()=>vi.advanceTimersByTime(1000));expect(r.output()).toBe('READY');r.click(start);act(()=>vi.advanceTimersByTime(500));r.click(action);expect(r.output()).toMatch(/PERFECT PARRY|S-RANK/);r.click(action);expect(r.output()).toBe('FAIL · NO ATTACK');});
it.each(['parry','dodge'])('%s offscreen stops attack; return cannot judge stale input',kind=>{const r=setup(kind);r.click(kind==='parry'?'ENEMY ATTACK':'ATTACK START');r.change(false,false);act(()=>vi.advanceTimersByTime(500));r.change(true,false);expect(r.output()).toBe('INTERRUPTED · RETRY');r.click(kind==='parry'?'PARRY':'DODGE LEFT');expect(r.output()).toBe('FAIL · NO ATTACK');r.unmount();expect(vi.getTimerCount()).toBe(0);});
it.each([0,439,440,929])('battle interruption at %sms has no stale state after return/reset',delay=>{const r=setup('battle-transition');r.click('ENCOUNTER');act(()=>vi.advanceTimersByTime(delay));r.change(false,false);r.change(true,false);expect(r.output()).toBe('FIELD · INTERRUPTED');r.click('RESET');act(()=>vi.advanceTimersByTime(2000));expect(r.output()).toBe('FIELD');expect(r.container.querySelector('.gameBattle')!.className).toContain('sceneFIELD');});
it('battle reduced switch cancels animation but completes logical transition; off -> new animation',()=>{const r=setup('battle-transition');r.click('ENCOUNTER');act(()=>vi.advanceTimersByTime(200));const animations=vi.mocked(HTMLElement.prototype.animate).mock.results.map(x=>x.value);r.change(true,true);expect(animations.every(a=>a.playState==='idle')).toBe(true);act(()=>vi.advanceTimersByTime(800));expect(r.output()).toBe('READY');expect(r.container.querySelector('.gameBattle')!.className).toContain('sceneBATTLE');r.click('RETURN');expect(r.output()).toBe('FIELD');r.change(true,false);r.click('ENCOUNTER');expect(vi.mocked(HTMLElement.prototype.animate).mock.calls).toHaveLength(2);r.click('RESET');act(()=>vi.advanceTimersByTime(2000));expect(r.output()).toBe('FIELD');});
it('battle continuous reverse and completed offscreen preserve final scene',()=>{const r=setup('battle-transition');r.click('ENCOUNTER');act(()=>vi.advanceTimersByTime(500));r.click('RETURN');act(()=>vi.advanceTimersByTime(1000));expect(r.output()).toBe('FIELD');r.click('ENCOUNTER');act(()=>vi.advanceTimersByTime(1000));r.change(false,false);r.change(true,false);expect(r.output()).toBe('READY');expect(r.container.querySelector('.gameBattle')!.className).toContain('sceneBATTLE');r.unmount();expect(vi.getTimerCount()).toBe(0);});
it.each(['running','paused','finished'])('combat %s canceled on reduction, reset and repeated input remain stable',state=>{const r=setup('combat');r.click('NORMAL');const old=vi.mocked(HTMLElement.prototype.animate).mock.results.map(x=>x.value);old.forEach(a=>a.playState=state);r.change(true,true);expect(old.every(a=>a.playState==='idle')).toBe(true);r.click('RESET');r.change(true,false);r.click('GUARD');expect(r.output()).toBe('GUARD -24');r.change(false,false);r.change(true,false);expect(r.output()).toBe('GUARD -24');r.click('RESET');expect(r.output()).toBe('READY');expect(vi.mocked(HTMLElement.prototype.animate).mock.results.every(x=>x.value.cancel.mock.calls.length)).toBe(true);r.unmount();expect(vi.getTimerCount()).toBe(0);});
